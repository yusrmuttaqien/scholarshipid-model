# Two-Tower Recommendation System — Model Building Guide

Step-by-step guide for building the two-tower neural network using TensorFlow/Keras Functional API.

---

## Prerequisites

- Python 3.10+
- TensorFlow 2.x
- pandas, numpy, scikit-learn
- Dataset generated by `generator_two_tower.py` → `datasets_two_tower/`

---

## Hyperparameters

| Parameter | Value | Rationale |
|-----------|-------|-----------|
| `EMBEDDING_DIM` | 64 | Sweet spot for 20K students / 800 scholarships |
| `HIDDEN_UNITS` | 128 → 64 | Student/Scholarship tower hidden layer sizes |
| `LEARNING_RATE` | 1e-3 | Adam default, with warmup + decay |
| `BATCH_SIZE` | 2048 | Large batch for CPU efficiency on M1 |
| `EPOCHS` | 50 (with EarlyStopping) | Patience=5, min_delta=0.001 |
| `LOSS` | WeightedMSE (custom) | Balanced gradient across 3 classes |
| `METRICS` | RMSE, MAE | Regression target evaluation |

### Language Proficiency Encoding

Parse the JSON list into fixed-size features:

```python
def encode_language_proficiency(df):
    """Extract max score per test type from language_proficiency JSON."""
    tests = ["toefl", "ielts", "topik", "jlpt", "delf", "hsk"]
    for test in tests:
        df[f"lang_{test}_score"] = 0.0
        df[f"has_{test}"] = False

    for idx, row in df.iterrows():
        for lp in row["language_proficiency"]:
            test = lp["test_type"]
            score = lp["score"]
            if test in tests:
                df.at[idx, f"lang_{test}_score"] = max(
                    df.at[idx, f"lang_{test}_score"], score
                )
                df.at[idx, f"has_{test}"] = True

    return df
```

## Section 5: Custom Components

### 5.1 CosineSimilarity Layer

Custom layer that computes cosine similarity between two embedding vectors and returns a scalar through sigmoid:

```python
class CosineSimilarity(layers.Layer):
    """Compute cosine similarity between two embeddings and apply sigmoid."""

    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def call(self, inputs):
        """inputs: [student_embedding, scholarship_embedding]"""
        student_emb, scholarship_emb = inputs

        # L2 normalize
        student_norm = tf.nn.l2_normalize(student_emb, axis=-1)
        schol_norm = tf.nn.l2_normalize(scholarship_emb, axis=-1)

        # Cosine similarity → scalar
        cosine_sim = tf.reduce_sum(
            tf.multiply(student_norm, schol_norm), axis=-1, keepdims=True
        )

        # Sigmoid to map [-1, 1] → [0, 1]
        return keras.activations.sigmoid(cosine_sim)

    def get_config(self):
        return super().get_config()
```

### 5.2 WeightedMSELoss

Custom loss function that applies class-aware weighting to prevent the model from ignoring minority classes:

```python
class WeightedMSE(keras.losses.Loss):
    """MSE loss with class-aware weighting for balanced gradient signal.

    Weights:
        - Match (>=0.7):      weight 1.0
        - In-Between (0.3-0.7): weight 1.5 (harder to predict)
        - Not Match (<0.3):   weight 1.0
    """

    def __init__(self, name="weighted_mse", **kwargs):
        super().__init__(name=name, **kwargs)

    def call(self, y_true, y_pred):
        """Compute weighted MSE."""
        mse = tf.square(y_true - y_pred)

        # Class weights
        in_between_mask = tf.logical_and(
            tf.greater(y_true, 0.3), tf.less(y_true, 0.7)
        )
        weights = tf.where(in_between_mask, 1.5, 1.0)

        return tf.reduce_mean(mse * weights)

    def get_config(self):
        return super().get_config()
```

### 5.3 ClassDistributionCallback

Custom callback that tracks per-class metrics during training:

```python
class ClassDistributionCallback(keras.callbacks.Callback):
    """Track RMSE and MAE per relevance class during training."""

    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.history = {
            "match_rmse": [], "in_between_rmse": [], "not_match_rmse": [],
            "match_mae": [], "in_between_mae": [], "not_match_mae": [],
        }

    def on_epoch_end(self, epoch, logs=None):
        if not hasattr(self, "_eval_model"):
            self._eval_model = self.model

        # Get predictions on validation set
        val_x_student = self.validation_data[0]["student_input"]
        val_x_scholarship = self.validation_data[0]["scholarship_input"]
        val_y = self.validation_data[1]

        predictions = self._eval_model(
            {"student_input": val_x_student, "scholarship_input": val_x_scholarship}
        ).numpy().flatten()

        # Class masks
        match_mask = val_y >= 0.7
        in_between_mask = (val_y >= 0.3) & (val_y < 0.7)
        not_match_mask = val_y < 0.3

        for name, mask in [("match", match_mask),
                            ("in_between", in_between_mask),
                            ("not_match", not_match_mask)]:
            if np.sum(mask) > 0:
                errors = predictions[mask] - val_y[mask]
                logs[f"{name}_rmse"] = np.sqrt(np.mean(errors ** 2))
                logs[f"{name}_mae"] = np.mean(np.abs(errors))
                self.history[f"{name}_rmse"].append(logs[f"{name}_rmse"])
                self.history[f"{name}_mae"].append(logs[f"{name}_mae"])

        print(f"\n  Epoch {epoch+1} per-class metrics:")
        for name in ["match", "in_between", "not_match"]:
            rmse = logs.get(f"{name}_rmse", 0.0)
            mae = logs.get(f"{name}_mae", 0.0)
            print(f"    {name}: RMSE={rmse:.4f}, MAE={mae:.4f}")
```

## Section 6: Two-Tower Architecture (Keras Functional API)

### Student Tower

```python
EMBEDDING_DIM = 64


def build_student_tower():
    """Build the student tower: inputs → embeddings → dense layers → embedding."""
    # Categorical inputs
    cat_inputs = {}
    cat_embeddings = []

    for name, cardinality in STUDENT_CATEGORICAL.items():
        inp = keras.Input(shape=(1,), name=f"stu_cat_{name}")
        cat_inputs[name] = inp
        emb_dim = EMBEDDING_DIMS.get(name, 8)
        x = layers.Embedding(input_dim=cardinality + 1, output_dim=emb_dim)(inp)
        x = layers.Flatten()(x)
        cat_embeddings.append(x)

    # Numerical inputs
    num_inp = keras.Input(
        shape=(len(STUDENT_NUMERICAL),), name="stu_numerical"
    )
    num_x = layers.Dense(32, activation="relu")(num_inp)

    # Boolean inputs
    bool_inp = keras.Input(
        shape=(len(STUDENT_BOOLEAN),), name="stu_boolean"
    )

    # Language inputs (6 test types × 2 features = 12)
    lang_inp = keras.Input(shape=(12,), name="stu_language")

    # Concatenate all features
    all_features = cat_embeddings + [num_x, bool_inp, lang_inp]
    concatenated = layers.Concatenate()(all_features)

    # Dense layers
    x = layers.Dense(128, activation="relu")(concatenated)
    x = layers.BatchNormalization()(x)
    x = layers.Dense(EMBEDDING_DIM, activation="relu")(x)
    x = layers.BatchNormalization()(x)

    return cat_inputs, num_inp, bool_inp, lang_inp, x


student_cat_inputs, student_num_inp, student_bool_inp, student_lang_inp, student_embedding = build_student_tower()
```

```python
def build_scholarship_tower():
    """Build the scholarship tower: inputs → embeddings → dense layers → embedding."""
    # Categorical inputs
    cat_inputs = {}
    cat_embeddings = []

    for name, cardinality in SCHOLARSHIP_CATEGORICAL.items():
        inp = keras.Input(shape=(1,), name=f"sch_cat_{name}")
        cat_inputs[name] = inp
        emb_dim = min(8, int(np.sqrt(cardinality)))
        x = layers.Embedding(input_dim=cardinality + 1, output_dim=emb_dim)(inp)
        x = layers.Flatten()(x)
        cat_embeddings.append(x)

    # Numerical inputs
    num_inp = keras.Input(
        shape=(len(SCHOLARSHIP_NUMERICAL),), name="sch_numerical"
    )
    num_x = layers.Dense(32, activation="relu")(num_inp)

    # Boolean inputs
    bool_inp = keras.Input(
        shape=(len(SCHOLARSHIP_BOOLEAN),), name="sch_boolean"
    )

    # List field embeddings (eligible_nationalities, eligible_fields, etc.)
    list_inp = keras.Input(shape=(32,), name="sch_list_fields")
    list_x = layers.Dense(16, activation="relu")(list_inp)

    # Concatenate all features
    all_features = cat_embeddings + [num_x, bool_inp, list_x]
    concatenated = layers.Concatenate()(all_features)

    # Dense layers
    x = layers.Dense(128, activation="relu")(concatenated)
    x = layers.BatchNormalization()(x)
    x = layers.Dense(EMBEDDING_DIM, activation="relu")(x)
    x = layers.BatchNormalization()(x)

    return cat_inputs, num_inp, bool_inp, list_inp, x


schol_cat_inputs, schol_num_inp, schol_bool_inp, schol_list_inp, scholarship_embedding = build_scholarship_tower()
```


### Build Feature Arrays

```python
def build_student_features(pairs_df, students_df):
    """Build student feature arrays from pairs."""
    merged = pairs_df.merge(students_df, on="student_id")

    # Categorical → integer encoding
    cat_arrays = {}
    for name in STUDENT_CATEGORICAL:
        lookup = {v: i for i, v in enumerate(pd.unique(merged[name].dropna()))}
        cat_arrays[name] = merged[name].map(lambda x: lookup.get(x, 0)).values.reshape(-1, 1)

    # Numerical → normalized
    num_arrays = merged[STUDENT_NUMERICAL].values.astype(np.float32)
    num_mean = num_arrays.mean(axis=0)
    num_std = num_arrays.std(axis=0) + 1e-8
    num_arrays = (num_arrays - num_mean) / num_std

    # Boolean → float
    bool_arrays = merged[STUDENT_BOOLEAN].astype(float).values.astype(np.float32)

    # Language features (12 dims: 6 tests × score + has_flag)
    lang_arrays = merged[[f"lang_{t}_score" for t in ["toefl","ielts","topik","jlpt","delf","hsk"]]
                          + [f"has_{t}" for t in ["toefl","ielts","topik","jlpt","delf","hsk"]]].values.astype(np.float32)

    return {
        "cat": cat_arrays,
        "num": num_arrays,
        "bool": bool_arrays,
        "lang": lang_arrays,
    }


def build_scholarship_features(pairs_df, scholarships_df):
    """Build scholarship feature arrays from pairs."""
    merged = pairs_df.merge(scholarships_df, on="scholarship_id")

    cat_arrays = {}
    for name in SCHOLARSHIP_CATEGORICAL:
        lookup = {v: i for i, v in enumerate(pd.unique(merged[name].dropna()))}
        cat_arrays[name] = merged[name].map(lambda x: lookup.get(x, 0)).values.reshape(-1, 1)

    num_arrays = merged[SCHOLARSHIP_NUMERICAL].values.astype(np.float32)
    num_mean = num_arrays.mean(axis=0)
    num_std = num_arrays.std(axis=0) + 1e-8
    num_arrays = (num_arrays - num_mean) / num_std

    bool_arrays = merged[SCHOLARSHIP_BOOLEAN].astype(float).values.astype(np.float32)

    # List fields → flattened encoding (32 dims)
    list_arrays = np.zeros((len(merged), 32), dtype=np.float32)
    for idx, row in merged.iterrows():
        vec = []
        for col in ["eligible_nationalities", "eligible_fields", "eligible_high_school_tracks"]:
            vals = row[col] if isinstance(row[col], list) else []
            vec.extend([1.0 if v else 0.0 for v in vals][:8])  # Max 8 per field
        list_arrays[idx, :len(vec)] = vec[:32]

    return {
        "cat": cat_arrays,
        "num": num_arrays,
        "bool": bool_arrays,
        "list": list_arrays,
    }
```

```python
                         [stu_batch["num"], stu_batch["bool"], stu_batch["lang"]],
        "scholarship_input": [sch_features_batch[n] for n in SCHOLARSHIP_CATEGORICAL] +
                             [sch_num_batch, sch_bool_batch, np.zeros((batch_size, 32))],
    }, verbose=0).flatten()

    for j, score in enumerate(scores):
        scholarship_scores.append({
            "scholarship_id": batch_sch["scholarship_id"].iloc[j],
            "name": batch_sch["name"].iloc[j],
            "predicted_score": float(score),
        })

# Top-K recommendations
results = pd.DataFrame(scholarship_scores).sort_values("predicted_score", ascending=False)
print("\nTop 10 recommendations:")
print(results.head(10).to_string(index=False))
```

### Optional: TF-IDF Text Similarity Bonus

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity as sk_cosine

def text_bonus(student_personal_statement, student_future_goals,
               scholarship_mission, scholarship_target_profile):
    """Simple TF-IDF text similarity bonus."""
    # Pair 1: personal_statement ↔ mission_statement
    vec1 = TfidfVectorizer(max_features=100)
    mat1 = vec1.fit_transform([student_personal_statement, scholarship_mission])
    sim1 = sk_cosine(mat1[0], mat1[1])[0][0]

    # Pair 2: future_goals ↔ target_recipient_profile
    vec2 = TfidfVectorizer(max_features=100)
    mat2 = vec2.fit_transform([student_future_goals, scholarship_target_profile])
    sim2 = sk_cosine(mat2[0], mat2[1])[0][0]

    # Bonus: 0.0 - 0.1 range
    return min(0.1, 0.05 * (sim1 + sim2))

# Apply text bonus to top results
sample_stu_text = students_df[students_df["student_id"] == sample_student_id].iloc[0]

results["text_bonus"] = results.apply(
    lambda row: text_bonus(
        sample_stu_text["personal_statement"],
        sample_stu_text["future_goals"],
        scholarships_df[scholarships_df["scholarship_id"] == row["scholarship_id"]]["mission_statement"].iloc[0],
        scholarships_df[scholarships_df["scholarship_id"] == row["scholarship_id"]]["target_recipient_profile"].iloc[0],
    ), axis=1
)

results["final_score"] = results["predicted_score"] + results["text_bonus"]
results = results.sort_values("final_score", ascending=False)

print("\nTop 5 with text bonus:")
```
